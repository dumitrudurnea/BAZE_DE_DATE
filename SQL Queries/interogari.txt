/*62Să presupunem că a fost mărită bursa cu 20%.
  Să se proiecteze o interogare la tabelul STUDENT,
  care ar afişa identificatorul studentului,
  numele lui şi bursa mărită. Datele afişate trebuie ordonate:
  a) după bursă; b) în ordinea alfabetică a numelui studenţilor.*/
SELECT st_id,
       nume,
       bursa * 1.2 AS BURSA_MARITA
FROM   student
ORDER  BY bursa_marita DESC,
          nume ASC;

/*63Să se formuleze o interogare care pentru fiecare student
  să se determine din tabelul NOTE_EXAM nota maximală
  şi nota minimală şi să le afişeze împreună
  cu identificatorul studentului.*/
SELECT st_id,
       Max(nota) AS nota_maxima,
       Min(nota) AS nota_minima
FROM   note_exam
GROUP  BY st_id;

/*64Să se formuleze o interogare care ar afişa lista
  cursurilor studiate afişate în ordinea descreşterii
  semestrelor şi în ordinea crescătoare a numărului
  de ore planifice pentru cursul respectiv.
  În tabelul rezultat cîmpul numărul semestrului trebuie
  să fie primul, după care denumirea şi identificatorul cursului.*/
SELECT semestru,
       curs_den,
       curs_id,
       ore
FROM   curs
WHERE  curs_id IN (SELECT DISTINCT note_exam.curs_id
                   FROM   note_exam)
ORDER  BY semestru DESC,
          ore ASC;

/*65Să se formuleze o interogare care ar afişa valoarea medie,
  minimală şi maximală a notelor studenţilor pentru fiecare
  dată de susţinerea a examenului şi prezintă rezultatele
  în ordinea descrescătoare a acestor valori.*/
SELECT data_exam,
       Avg(nota) AS nota_medie,
       Min(nota) AS nota_minima,
       Max(nota) AS nota_maxima
FROM   note_exam
GROUP  BY data_exam
ORDER  BY nota_medie DESC,
          nota_minima ASC,
          nota_maxima ASC;

       INNER JOIN note_exam n

                 FROM   note_exam

              ON s.univ_id = u.univ_id

/*93Să se formuleze o interogare care ar afişa  numele studenţilor cu acelaşi prenume.*/

       s1.prenume,
       s2.nume,
       s2.prenume
FROM   student s1
       JOIN student s2
         ON s1.prenume = s2.prenume
            AND s1.nume != s2.nume

          s1.nume;

/*94Să se formuleze o interogare care ar afişa toate perechile de nume a studenţilor care locuiesc în acelaşi oraş, cu excepţia combinaţiei numelui studenţilor cu ei înseşi (Popescu – Popescu) şi a combinaţiilor ce diferă după ordinea numelor (Popescu – Ionescu sau Ionescu – Popescu).
*/
SELECT DISTINCT s1.nume,
                s1.prenume,
                s2.nume,
                s2.prenume
FROM   student s1,
       student s2
WHERE  s1.oras = s2.oras
       AND s1.st_id < s2.st_id
       AND s1.nume < s2.nume;

/*95Să se formuleze o interogare care ar afişa lista tuturor perechilor de denumiri de universităţi care se află în acelaşi oraş.*/
SELECT DISTINCT U1.univ_den AS universitate1,
                U2.univ_den AS universitate2
FROM   universitate U1,
       universitate U2
WHERE  U1.oras = U2.oras
       AND U1.univ_den < U2.univ_den
ORDER  BY U1.univ_den,
          U2.univ_den;
